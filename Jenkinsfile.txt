pipeline {
    agent any

    stages {
        stage('Clone Repository') {
            steps {
                git 'https://github.com/your-username/jenkins-java-demo.git'
            }
        }

        stage('Compile') {
            steps {
                bat 'javac App.java'
            }
        }

        stage('Run') {
            steps {
                bat 'java App'
            }
        }
    }
}
